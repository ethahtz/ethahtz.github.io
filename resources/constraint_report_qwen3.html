def extract_protagonist_gender(ctx: dict) -> str | None:
    # Strip dialogue so other speakers' pronouns don't pollute the count.
    head = re.sub(r'["“][^"”]*["”]', "", ctx["body"][:1500])
    m = len(_PRON_MALE.findall(head))
    f = len(_PRON_FEMALE.findall(head))
    has_protagonist = bool(ctx["ner_persons"]) or bool(
        re.search(r"\b(named|called)\s+[A-Z][a-z]+", ctx["body"])
    )
    if not has_protagonist and m + f == 0:
        return None
    if m + f < 2:
        return None
    if m >= 2 * f and m >= 3:
        return "male"
    if f >= 2 * m and f >= 3:
        return "female"
    return None

async def llm_judge_async(
    spec: dict[str, Any],
    story: str,
    client: Any | None,
    model: str = DEFAULT_LLM_JUDGE_MODEL,
    max_retries: int = 3,
    backoff_base: float = 1.5,
) -> Any:
    if client is None:
        return None
    prompt = _llm_judge_prompt(spec, story)
    last_err: Exception | None = None
    for attempt in range(max_retries):
        try:
            resp = await client.chat.completions.create(
                model=model,
                messages=[{"role": "user", "content": prompt}],
                temperature=0.0,
            )
            break
        except Exception as e:  # noqa: BLE001
            last_err = e
            if attempt == max_retries - 1:
                return {"error": f"{type(e).__name__}: {e}"}
            await asyncio.sleep(backoff_base ** attempt)
    else:
        return {"error": f"{type(last_err).__name__}: {last_err}"}
    content = (resp.choices[0].message.content or "").strip()
    content = re.sub(r"^```(?:json)?\s*|\s*```$", "", content, flags=re.IGNORECASE)
    try:
        return json.loads(content).get("value")
    except json.JSONDecodeError:
        return content.strip('"').strip()

def extract_protagonist_occupation(ctx: dict) -> str:
    body = ctx["body_lower"]
    counts: Counter[str] = Counter()
    for option, forms in _OCCUPATION_LEXICON.items():
        for sf in forms:

async def llm_judge_async(
    spec: dict[str, Any],
    story: str,
    client: Any | None,
    model: str = DEFAULT_LLM_JUDGE_MODEL,
    max_retries: int = 3,
    backoff_base: float = 1.5,
) -> Any:
    if client is None:
        return None
    prompt = _llm_judge_prompt(spec, story)
    last_err: Exception | None = None
    for attempt in range(max_retries):
        try:
            resp = await client.chat.completions.create(
                model=model,
                messages=[{"role": "user", "content": prompt}],
                temperature=0.0,
            )
            break
        except Exception as e:  # noqa: BLE001
            last_err = e
            if attempt == max_retries - 1:
                return {"error": f"{type(e).__name__}: {e}"}
            await asyncio.sleep(backoff_base ** attempt)
    else:
        return {"error": f"{type(last_err).__name__}: {last_err}"}
    content = (resp.choices[0].message.content or "").strip()
    content = re.sub(r"^```(?:json)?\s*|\s*```$", "", content, flags=re.IGNORECASE)
    try:
        return json.loads(content).get("value")
    except json.JSONDecodeError:
        return content.strip('"').strip()
